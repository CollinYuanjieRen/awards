def Erdos210.D : ℝ × ℝ → ℝ × ℝ → ℝ × ℝ → ℝ :=
fun a b c => (b.1 - a.1) * (c.2 - a.2) - (b.2 - a.2) * (c.1 - a.1)
def Erdos210.OnLine : ℝ × ℝ → ℝ × ℝ → ℝ × ℝ → Prop :=
fun a b z => D a b z = 0
def Erdos210.ordinaryLines : Finset (ℝ × ℝ) → Set (AffineSubspace ℝ (ℝ × ℝ)) :=
fun P => {L | ∃ a ∈ P, ∃ b ∈ P, a ≠ b ∧ L = line[ℝ, a, b] ∧ {x ∈ P | x ∈ L}.card = 2}
def Erdos210.f : ℕ → ℕ :=
