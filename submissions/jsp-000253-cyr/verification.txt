fun A => ∃ a ∈ A, ∃ B ⊆ A, a ∉ B ∧ B.Nonempty ∧ (↑a)⁻¹ = ∑ b ∈ B, (↑b)⁻¹

fun A => ∃ a ∈ A, ∃ b ∈ A, a ≠ b ∧ ∃ k, 0 < k ∧ (↑a)⁻¹ = ↑k * (↑b)⁻¹
