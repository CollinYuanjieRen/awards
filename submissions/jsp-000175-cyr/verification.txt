def Erdos181.hypercube : (n : ℕ) → SimpleGraph (Fin n → Bool) :=
fun n => fromRel fun u v => ∃ i, u i ≠ v i ∧ ∀ (j : Fin n), j ≠ i → u j = v j
def Erdos181.GraphRamseyProp.{u_1} : {V : Type u_1} → ℕ → SimpleGraph V → Prop :=
fun {V} N G => ∀ (R : SimpleGraph (Fin N)), G ⊑ R ∨ G ⊑ Rᶜ
def Erdos181.graphRamsey.{u_1} : {V : Type u_1} → SimpleGraph V → ℕ :=
fun {V} G => sInf {N | GraphRamseyProp N G}
def Erdos181.IsGood.{u_1} : {V : Type u_1} → SimpleGraph V → ℕ → ℕ → Prop :=
fun {V} G m n => G.CliqueFree m ∧ Gᶜ.CliqueFree n
def Erdos181.RamseyProp : ℕ → ℕ → ℕ → Prop :=
fun N m n => ∀ (G : SimpleGraph (Fin N)), ¬IsGood G m n
def Erdos181.ramsey : ℕ → ℕ → ℕ :=
fun m n => sInf {N | RamseyProp N m n}
def Erdos181.R : ℕ → ℕ :=
fun n => ramsey n n
